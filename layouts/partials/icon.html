{{/* Ícones SVG inline. Substitui o Font Awesome (100 KB de CSS + webfonts no CDN) por ~1 KB de markup: só 6 ícones são usados em todo o site. Uso: {{ partial "icon.html" (dict "name" "github") }} {{ partial "icon.html" (dict "name" "sun" "size" 16 "class" "ico-light") }} Nome inválido quebra o build (errorf) em vez de renderizar nada — um ícone invisível passa despercebido, um build vermelho não. */}} {{- $name := .name -}} {{- $size := .size | default 18 -}} {{- $class := .class | default "" -}} {{- $paths := dict "github" `` "linkedin" `` "rss" `` "sun" `` "moon" `` "half" `` -}} {{- with index $paths $name -}} {{- else -}} {{- errorf "partial icon.html: ícone %q não existe (disponíveis: github, half, linkedin, moon, rss, sun)" $name -}} {{- end -}}
